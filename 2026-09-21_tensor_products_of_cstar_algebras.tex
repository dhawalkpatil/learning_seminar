\section{Cross Norms}
\begin{notation}
In this section, $A_1,$ $A_2$ will denote Banach spaces.
The norms on these spaces will be denoted by $\lVert \cdot \rVert$
unambiguously as we will distinguish between elements of $A_1$ and $A_2.$
\end{notation}
\begin{notation}[]
For Banach spaces $A_1, A_2,$ we will denote their \emph{algebraic tensor product} by $A_1 \otimes A_2.$
Its completion in some norm $\lVert \cdot \rVert_\beta$ will be denoted by $A_1 \otimes_\beta A_2.$
\end{notation}
\begin{definition}
	A norm $\lVert \cdot \rVert_c$ on $A_1 \otimes A_2$
is called a \emph{cross norm} if it satisfies $\lVert x_1 \otimes x_2 \rVert_c = \lVert x_1 \rVert \lVert x_2 \rVert$
for all $x_1 \in A_1, x_2 \in A_2.$
\end{definition}
\begin{definition}
	The \emph{injective cross norm} $\lVert \cdot \rVert_\lambda$ on $A_1 \otimes A_2$ is defined by
	$$  \lVert \xi \rVert = \sup \{ \lvert\sum_{k=1}^{n} f(x_{1,k}) g(x_{2,k}) \rvert 
		\colon 
		\begin{array}{c}
		\xi = \sum_{k=1}^{n} x_{1,k}\otimes x_{2,k} = \xi, \\
		f \in A_1^\ast, \lVert f \rVert = 1, \\
		g \in A_2^\ast, \lVert g \rVert = 1
	\end{array}
\}.$$
\end{definition}
\begin{remark}
	That $\lVert \cdot \rVert_\lambda$ is a cross norm is immediate from an application of the Hahn Banach Theorem.
\end{remark}

\begin{remark}
	Let $\lVert \cdot \rVert_\beta$ be a cross norm on $A_1 \otimes A_2.$ 
	Retaining the notation of the previous definition, 
	$$ \lvert f(x_1) \rvert \lvert g(x_2) \rvert
	\leq \lVert x_1 \rVert \lVert x_2 \rVert = \lVert x_1 \otimes x_2 \rVert_\beta.$$
\end{remark}
So, the injective cross norm is the smallest cross norm on $A_1 \otimes A_2.$

\begin{definition}
	The \emph{projective cross norm} on $A_1 \otimes A_2$ is denoted by 
	$\lVert \cdot \rVert_\gamma$ and defined as
	$$\lVert \xi \rVert = \inf \{ \sum_{k=1}^{n} \lVert x_{1,k} \rVert \lVert x_{2,k} \rVert
	\colon \xi = \sum_{k=1}^{n} x_{1,k} \otimes x_{2,k} \}.$$
\end{definition}

\begin{remark}
	Let $\lVert \cdot \rVert_\beta$ be a cross norm on $A_1 \otimes A_2.$ 
	For $\xi$ in $A_1 \otimes A_2,$ and $\epsilon>0,$ there exists a representation
	$\xi = \sum_{k=1}^{n} x_{1,k} \otimes x_{2,k}$ in $A_1 \otimes A_2,$ such that
	\[
		\lVert \xi \rVert_\beta \leq
		\sum_{k=1}^{n} \lVert x_{1,k} \otimes x_{2,k} \rVert_\beta
		= \sum_{k=1}^{n} \lVert x_{1,k} \rVert \lVert x_{2,k} \rVert
		\leq \lVert \xi \rVert_\gamma + \epsilon.
	\]
\end{remark}
So, the projective norm majorises all cross norms. 
That it is itself a cross norm is evident from the computation
$$ \lVert x_1 \rVert \lVert x_2 \rVert = 
\lVert x_1 \otimes x_2 \rVert_\gamma \leq \lVert x_1 \otimes x_2 \rVert_\lambda \leq \lVert x_1 \rVert \lVert x_2 \rVert.$$
The same computation reveals that all norms `between' the two norms are cross norms.

\section{$C^\ast$ cross-norms}
The $C^\ast$ norms on $A_1 \otimes A_2$ can always be extended to unitisations.
\begin{theorem}
    Let $A_1$ and $A_2$ be $\text{C}^\ast$\nobreakdash-algebras with a $C^\ast$ norm $\lVert \cdot \rVert_\beta$ on 
    $A_1 \otimes A_2.$ There exists a $\text{C}^\ast$ norm $\lVert \cdot \rVert_{\beta_0}$ on $(A_1)_I \otimes (A_2)_I$
    which extends $\lVert \cdot \rVert_{\beta}.$
\end{theorem}
\begin{proof}
	Let $\pi$ be a faithful represntation of $A_1 \otimes_\beta A_2.$ Then there exist representations
	$\pi_i$ of $A_i$ on the same Hilbert Space $\mathcal{H}$ satisfying
	\[
		\pi(x_1 \otimes x_2) = \pi_1(x_1) \pi_2(x_2) = \pi(x_2) \pi_1(x_1)
	\]
	for all $x_1$ in $A_1,$ all $x_2$ in $A_2$ and
	$\pi_1(x_1) = \lim_{\lambda} \pi(x_1 \otimes u_{\lambda})$ for an approximate uni $\{ u_{\lambda} \}$ of $A_1.$
	Since $\pi$ is faithful, $\lVert  x_1 \otimes x_2 \rVert_{\beta} = \lVert \pi(x_1 \otimes x_2) \rVert.$
	Define $\pi_i^0 (x, \lambda) = \pi_i(x_i) + \lambda$ on $(A_i)_I.$ Since the ranges of $\pi_1^0$ and $\pi_2^0$ 
	commute with each other, $\pi_0(x_1 \otimes x_2) = \pi_1^0(x_1) \pi_2^0(x_2)$ defines a valid representation
	of $(A_1)_I \otimes (A_2)_I.$ Define $\lVert x \rVert_{\beta_0} = \lVert \pi_0(x) \rVert.$
	This norm extends $\lVert \cdot \rVert_{\beta}.$
\end{proof}
In general, the projective cross\nobreakdash-norm is not a $C^\ast$~norm \cite[Remark 4.4]{Pisier_2020}.
So we consider 
\begin{definition}
	The maximal $\text{C}^\ast$ norm defined by
	\[
		\lVert x \rVert_{\max} = \sup \{
		\lVert \pi(x) \rVert \colon \pi \text{ is a representation of } A_1 \otimes A_2 \}
	\]

\end{definition}
The max norm is a cross norm.
\begin{remark}
	For $\xi = \sum_{k=1}^{n} x_{1,k} \otimes x_{2,k},$ in $A_1 \otimes A_2,$ we have
	$$ \lVert \pi(\xi) \rVert
	= \lVert \sum_{k=1}^{n} \pi(x_{1,k} \otimes x_{2,k}) \rVert
	\leq \sum_{k=1}^{n} \lVert \pi(x_{1,k} \otimes x_{2,k}) \rVert
	\leq \sum_{k=1}^{n} \lVert x_{1,k} \otimes x_{2,k} \rVert .$$
	This shows that $\lVert x \rVert_{\max} \leq \left\lVert x \right\rVert_\gamma.$

	In addition, for any representation $\pi_1$ of $A_1$ and $\pi_2$ of $A_2,$ we have 
	\begin{flalign*}
		\lVert \pi_1(x_1) \rVert \lVert \pi_2(x_2) \rVert 
		=\lVert \pi_1(x_1) \pi_2(x_2)\rVert
		= \lVert (\pi_1 \otimes \pi_2) (x_1 \otimes x_2) \rVert \\
		\leq \lVert x_1 \otimes x_2 \rVert_{\max} 
		\leq \lVert x_1 \otimes x_2 \rVert_\gamma
		\leq \lVert x_1 \rVert \lVert x_2 \rVert.
	\end{flalign*}
	By choosing faithful representations $\pi_1$ and $\pi_2,$ the left hand side and right hand side become equal.
\end{remark}

\begin{definition}
	The minimal $C^\ast$~cross\nobreakdash-norm $\lVert \cdot \rVert_{\min}$ is defined by
$$\lVert \xi \rVert_{\min} 
= \sup\{ \lVert (\pi_1 \otimes \pi_2) (\xi) \rVert \colon 
\begin{array}{c}
\pi_1 \text{ is a representation of } A_1, \\
\pi_2 \text{ is a representation of } A_2 
\end{array}
\} .$$
\end{definition}
It is immediate that $\lVert \cdot \rVert_{\min} \leq \lVert \cdot \rVert_{\max} \leq 
\lVert \cdot \rVert_\gamma.$ 
\begin{remark}
	To see that $\lVert \cdot \rVert_{\min}$ is a cross norm, we only need to show that $\lVert \cdot \rVert_{\lambda}
	\leq \lVert \cdot \rVert_{\min}.$
	For $f$ in $A_1^\ast,$ there exist a representation $\pi_i$ of $A_1$ on a Hilbert Space $\mathcal{H}_1 $ and $\xi_1, \eta_1$
	in $\mathcal{H}_1$ satisfying $\lVert f \rVert = \lVert \xi_1 \rVert \lVert \eta_1 \rVert$ and
	$f(a) = \langle\eta_1, \pi_1(a) \xi_1\rangle$ for all $a$ in $A_1.$ Similarly, for $g$ in $A_2^\ast,$ there exist a representation
	$\pi_2$ of $A_2$ on a Hilbert space $\mathcal{H}_2$ and $\xi_2, \eta_2$ in $\mathcal{H}_2$ such that
	$g(b) = \langle \eta_2, \pi_2(b) \xi_2 \rangle$ for all $b$ in $A_2$ and $\lVert g \rVert = \lVert \xi_2 \rVert \lVert \eta_2 \rVert.$ This means for $x = \sum_{k=1}^{n} x_{1,k} \otimes x_{2,k}$ in $A_1 \otimes A_2,$ we have
	\begin{align*}
		{}& \lvert (f \otimes g) x \rvert \\
		={}&  \lvert \sum_{n=1}^{k} f(x_{1,k}) g(x_{2,k}) \rvert \\
		\leq{}&  \lvert \sum_{n=1}^{k} f(x_{1,k}) g(x_{2,k}) \rvert \\
		\leq{}&  \lvert \sum_{k=1}^{n} \langle\eta_1, \pi_1(x_{1,k}) \xi_1\rangle \rvert
		\lvert \langle\xi, \pi_2(x_{2,k}) \xi_2\rangle \rvert \\
		={}& \lvert \sum_{k=1}^{n} \langle\eta_1, \pi_1(x_{1,k}) \xi_1\rangle 
		\langle\xi, \pi_2(x_{2,k}) \xi_2\rangle \rvert \\
		={}& \lvert \sum_{n=1}^{k} \langle \eta_1 \otimes \eta_2, (\pi_1 \otimes \pi_2) (x_{1,k} \otimes x_{2,k}) (\xi_1 \otimes \xi_2) \rangle \rvert \\
		={}&  \lvert \langle \eta_1 \otimes \eta_2, (\pi_1 \otimes \pi_2) ( x) (\xi_1 \otimes \xi_2)\rangle \rvert \\
		\leq{}& \lVert \eta_1 \rVert \lVert \eta_2 \rVert \lVert (\pi_1 \otimes \pi_2)(x) \rVert
		\lVert \xi_1 \rVert \lVert \xi_2 \rVert \\
		={}& \lVert f \rVert \lVert g \rVert \lVert x \rVert_{\min}
	\end{align*}
	In other words, $\lVert x \rVert_\gamma \leq \lVert x \rVert_{\min}.$
\end{remark}
Thus, $\lVert \cdot \rVert_{\min}$ is a cross norm.

\section{Universal Property}
\begin{theorem}
    For representations $\pi_1,$ respectively $\pi_2,$ of $A_1,$ respectively $A_2,$ onto a Hilbert Space $\mathcal{H}$
    such that the images of $\pi_1$ and $\pi_2$ commute, there exists a unique representation $\pi$ of $A_1 \otimes_{\max}
    A_2$ on the same space $\mathcal{H}$ such that
    $$\pi(x_1 \otimes x_2) = \pi_1(x_1) \pi_2(x_2)$$
    and the image of $\pi$ is the $\text{C}^\ast$\nobreakdash-algebra generated by the images of $\pi_1$ and $\pi_2.$
\end{theorem}
\begin{proof}
	Define $\pi(x_1 \otimes x_2) = \pi_1(x_1) \pi_2(x_2)$ for $x_1$ in $A_1,$ and $x_2$ in $A_2.$
	For $x = \sum_{k=1}^{n} x_{1,k} \otimes x_{2,k}$ and $y = \sum_{l=1}^{m} y_{1,l} \otimes y_{2,l}$
	in $A_1 \otimes A_2,$ we have
	\begin{align*}
		{}& (\pi_1(x_{1,k}) \otimes \pi_2(x_{2,k})) (\pi_1(y_{1,l}) \otimes \pi_2(y_{2,l})) \\
		={}& \pi_1(x_{1,k}) \pi_2(_{2,k}) \pi_1(y_{1,l}) \pi_2(y_{2,l}) \\
		={}& \pi_1(x_{1,k}) \pi_1(y_{1,l}) \pi_2(_{2,k}) \pi_2(y_{2,l}) \\
		={}& \pi_1(x_{1k} y_{1,l}) \pi_2(x_{2,k} y_{2,l}),
	\end{align*}
	for every $k$ in $\{1,2,\dotsc, n\}$ and $l$ in $\{1, 2, \dotsc, m\}.$
	Taking linear combinations over $k,l$ we get
	$$\pi(x) \pi(y) = \pi(xy).$$

	Also, $\lVert (\pi_1 \otimes \pi_2) (\sum_{k=1}^{n} x_{1,k} \otimes x_{2,k} \rVert) \leq 
	\lVert \sum_{k=1}^{n} x_{1,k} \otimes x_{2,k} \rVert_{\max}.$ So $\pi$ is norm decreasing. Hence there
	exists an extension of $\pi$ to all of $A_1 \otimes_{\max} A_2.$

Moreover, any other representation $\rho$ which satisfies $\rho(x_1 \otimes x_2) = \pi_1(x_1) \otimes \pi_2(x_2)$ must agree
with $\pi$ on the completion because they agree on a dense subset.
\end{proof}

\begin{remark}
	This shows that $A_1 \otimes_{\max} A_2$ is embedded inside $A_1 \otimes_\beta A_2$ for every $C^\ast$~cross\nobreakdash-norm $\beta.$
\end{remark}


\begin{theorem}
    Let $A_1, A_2$ be $\text{C}^\ast$\nobreakdash-algebras and denote $A_1 \otimes_{\min} A_2$ by $A_{\min}$ and $A_1 \otimes_{\max} A_2$
    by $A_{\max}.$ Then 
\begin{enumerate}
	\item If $\omega_1 \in (A_1^\ast)_+,$ and
    $\omega_2 \in (A_2^\ast)_+,$ then $\omega = \omega_1 \otimes \omega_2$ is in $((A_{\max}^\ast)_+$
    and induces the representation $(\pi, \mathcal{H}, \xi)$ which is the tensor product of the representations
    $(\pi_1, \mathcal{H}_1, \xi_1)$ and $(\pi_2, \mathcal{H}_2, \xi_2).$
\item $$\lVert x \rVert_{\min} = 
	\sup \left\{ \frac{(\omega_1 \otimes \omega_2) (a^\ast x^\ast x a)^{1/2} }
		{ (\omega_1 \otimes \omega_2)(a^\ast a)^{1/2} } 
\begin{array}{c}
	\colon \omega_1 \in (A_1^\ast)_+, \\
	\omega_2 \in (A_2^\ast)_+ ,\\
	a \in A_{\min}
\end{array}
\right\} $$
\end{enumerate}
\end{theorem}

\begin{proof}
	Define $\mathcal{H} = \mathcal{H}_1 \otimes \mathcal{H}_2,$ $\xi = \xi_1 \otimes \xi_2$ and $\pi = \pi_1 \otimes
	\pi_2.$ Then,
\begin{align*}
	& \langle\xi, \pi(x_1 \otimes x_2) \xi\rangle \\
	={}& \langle\xi_1 \otimes \xi_2, ( \pi_1(x_1) \otimes \pi_2(x_2) ) (\xi_1 \otimes \xi_2) \rangle \\
	={}& \omega_1(x_1) \omega_2(x_2) \\
	={}& (\omega_1 \otimes \omega_2) (x_1 \otimes x_2) .
\end{align*}
Taking finite linear combinations, we get that $\omega(x) = \langle \xi, \pi(x) \xi\rangle$ for all $x$ on the algebraic
tensor product $A_1 \otimes A_2.$ This shows that $\omega$ is positive in $A_{\max}.$ Also, $\pi_1(A_1) \xi_1$
is dense in $\mathcal{H}_1,$ $\pi_2(A_2) \xi_2$ is dense in $\mathcal{H}_2.$ So their tensor product $(\pi_1(A_1) \xi_1 )
\otimes (\pi_2(A_2) \xi_2) = \pi(A_1 \otimes A_2) \xi$ is dense in $\mathcal{H}.$ Thus, $\xi$ is cyclic under $\pi.$

For the second assertion, pick $x$ in $A_{\min}.$ Given representations $\pi_1,$ respectively $\pi_2,$
of $A_1,$ respectively $A_2,$ we can express them as direct sums of irreducible representations as
$\pi_1 = \oplus_i \pi_{1,i}$ and $\pi_2 = \oplus_{j} \pi_{2,j}.$
Then
$$\lVert (\pi_1 \otimes \pi_2)(x) \rVert
= \sup_{i,j} \{ (\pi_{1,j} \otimes \pi_{2,j} ) (x) \}.$$
Since $\lVert \oplus_{j} a_i \rVert = \sup \{ \lVert a_i \rVert \colon i \},$
we can rewrite
\begin{align*}
	{} \lVert x \rVert_{\min} 
	={} \sup \{ \lVert (\pi_1 \otimes \pi_2) (x) \rVert \colon
\pi \text{ is a cyclic representation of } A_1 \otimes A_2. \}
\end{align*}
Let $\xi_i$ be the cyclic vector for $\pi_i,$ $i=1,2.$
Replacing $\pi_1,$ respectively $\pi_2$ with $\omega_1, \in (A_1^\ast)_+,$ respectively $\omega_2
\in (A_2^\ast)_+,$ we get
\begin{align*}
	& \lVert (\pi_1 \otimes \pi_2)(x) \rVert \\
	={}& \lVert (\pi_1 \otimes \pi_2)(x)^\ast (\pi_1 \otimes \pi_2)(x) \rVert^{1/2} \\
	={}& \sup\{ \langle \eta, (\pi_1 \otimes \pi_2) ( x^\ast x) \eta\rangle \colon \eta \not = 0 \} \\
	={}& \sup\{ \frac{ \langle(\pi_1 \otimes \pi_2) (a) \xi, (\pi_1 \otimes \pi_2)(x^\ast x) (\pi_1 \otimes \pi_2)(a) \xi \rangle }
	{\langle (\pi_1 \otimes \pi_2) (a) \xi, (\pi_1 \otimes \pi_2)(a) \xi \rangle} \} \\
	={}& \sup \{ \frac{\langle \xi, (\pi_1 \otimes \pi_2) ( a^\ast x^\ast x a) \xi \rangle}
	{\langle\xi, (\pi_1 \otimes \pi_2)(a^\ast a) \xi \rangle} \colon a \not = 0, a \in A_1 \otimes A_2 \} \\
	={}& \sup \{ \frac{(\omega_1 \otimes \omega_2) ( a^\ast x^\ast x a) }
	{(\omega_1 \otimes \omega_2)(x^\ast x)} \colon a \in A_1 \otimes A_2 \}
\end{align*}
\end{proof}


\section{Takesaki's Theorem}

We will start this section with a lemma about product states, and 
consider a subset $S_\beta$ of the continuous product pure states.
Then we will prove that the injective cross norm and the injective
cstar norm coincide if one of the factors is abelian. This proof makes
use of an identification of $A_1 \otimes A_2$ with a subspace
of the continuous $A_1$ valued functions on the character space of $A_2.$
This theorem will be useful in proving that the all cstar norms are
equal if one of the factors is abelian. Finally we will show that
the injective cstar norm is the minimal cstar norm, hence all
cstar norms are cross norms.

\begin{proposition}
	If $\omega_1$ is a positive linear functional on $A_1$, and 
	$\omega_2$ is a positive linear functional on $A_2,$ then $\omega_1 \otimes
	\omega_2$ is positive on $A_1 \otimes A_2.$
\end{proposition}
\begin{proof}
	Suppose $x = \sum_{i=1}^{n} x_{1,i} \otimes x_{2,i}$ is in $A_1 \otimes
	A_2.$ Then,
\begin{align*}
	{}& (\omega_1 \otimes \omega_2) (x^\ast x) \\
	={}& (\omega_1 \otimes \omega_2) (\sum_{i,j=1}^{n} 
	(x_{1,i} \otimes x_{2,i} )^\ast \otimes (x_{1,i} \otimes x_{2,i} ) \\
	={}& (\omega_1 \otimes \omega_2) (\sum_{i,j=1}^{n} 
	(x_{1,i}^\ast x_{1,i} \otimes x_{2,i}^\ast \otimes x_{2,i} ) \\
	={}& \sum_{i,j=1}^{n} 
	\omega_1 (x_{1,i}^\ast x_{1,i}) \omega_2 (x_{2,i}^\ast x_{2,i}).
\end{align*}
For complex numbers $\lambda_1, \dotsc, \lambda_n,$ we have
\begin{align*}
	{}& \sum_{i,j=1}^{n} \overline{\lambda_i} \lambda_j \omega_2 (x_{2,i}^\ast x_{2,j} )  \\
{}&= \omega_2( \sum_{i,j=1}^{n} \overline{\lambda_i} \lambda_j x_{2,i}^\ast x_{2,j} ) \\
{}&= \omega_2( ( \sum_{i=1}^{n} {\lambda_i}  x_{2,i})^\ast ( \sum_{j=1}^{n} \lambda_j x_{2,j}) )  \geq 0.
\end{align*}
This means that the matrix $A = [\omega_2(x_{2,i} ^\ast x_{2,j} ]$ is positive semidefinite and has a representation $A = \sum_{l=1}^{n} [\overline{r_{i,l}} r_{j,l} ],$
where $[\overline{r_{l,i}} r_{l,j} ]$ is a scaling of some rank one projection 
onto the subspace spanned by $(r_{l,1}, \dotsc, r_{l,n} ).$  
This means
\begin{align*}
	{}& (\omega_1 \otimes \omega_2) (x^\ast x) \\
	={}&  \sum_{i,j=1}^{n} \omega_1(x_{1,i}^\ast x_{2,j}) \sum_{l=1}^{n} \overline{r_{l,i}} r_{l,j} \\
	={}& \sum_{l=1}^{n} \omega_1(\sum_{i,j=1}^{n} \overline{r_{l,i}} r_{l,j} 
	x_{1,i}^\ast x_{2,j} ) \\
	={}& \sum_{l=1}^{n} \omega_1( (\sum_{i=1}^{n} r_{l,i} x_{1,i} )^\ast
	( \sum_{j=1}^{n} r_{l,j} x_{2,j})) \geq 0.
\end{align*}

\end{proof}

\begin{notation}
	The set of all Pure states on $A_1$ is denoted by $P(A_1).$
\end{notation}

\begin{notation}
	Given a  cstar norm $\beta$ on $A_1 \otimes A_2$,
	we denote
	$$ S_ \beta = \{ ( \omega_1 \otimes \omega_2) \colon
		\omega_1 \otimes \omega_2 \text{ is continuous in the }
	\beta \text{ norm }. \} $$
\end{notation}

\begin{lemma}
	The set $S_\beta$ is a closed subset of $P(A_1) \times P(A_2)$
	with the product $\text{weak}^\ast$ topology.
\end{lemma}

\begin{proof}
	Let $\omega_{1,\alpha} \otimes \omega_{2,\alpha}$ be a net in $S_\beta$
	such that $\lim_{\alpha} \omega_{1,\alpha} (x_1) = \omega_{1} (x_1)$
	for each $x_1$ in $A_1,$ and $\lim_{\alpha} \omega_{2,\alpha} (x_2)
	= \omega_{2} (x)2$ for each $x_2$ in $A_2.$
	For a fixed $x = \sum_{l=1}^{k} x_{1,l} \otimes x_{2,l}$ in $A_1 \otimes A_2,$ 
\begin{align*}
	\omega_1 \otimes \omega_2 (x)
	={}& \omega_1 \otimes \omega_2 
	(\sum_{l=1}^{k} x_{1,l} \otimes x_{2,l}) \\
	={}& \sum_{l=1}^{k} \omega_1(x_{1,l}) \omega_2(x_{2,l}) \\
	={}& \sum_{l=1}^{k} \lim_i \omega_{1,\alpha} (x_{1,l}) \omega_{2,\alpha} \lim_i (x_{2,l}) \\
	={}& \sum_{l=1}^{k} \lim_i (\omega_{1,\alpha} \otimes \omega_{2,\alpha} )
	(x_{1,l} \otimes x_{2,l}) \\
	={}& \lim_{\alpha} (\omega_{1,\alpha} \otimes \omega_{2,\alpha} ) (x) \\
\end{align*}
For $x$ in $A_1 \otimes A_2,$ we have $\lvert (\omega_{1,\alpha} \otimes \omega_{2,\alpha}) (x) \rvert \leq \lVert x \rVert_{\beta}$ for each $i.$ So, $\lvert \omega_1 \otimes \omega_2 (x) \rvert \leq \lVert x \rVert_{\beta}.$
This means $\omega_1 \otimes \omega_2$ is also in $S_\beta.$
\end{proof}

\begin{theorem}
    If $A_2$ is abelian, then the $\min$ norm and the $\lambda$ norm coincide.
\end{theorem}
\begin{proof}
	Let $A_2$ be $C(\Omega)$ for some compact Hausdorff Space $\Omega.$
	For
	\[
		x = \sum_{i=1}^{n} x_{1,i} \otimes x_{2,i}
	\]
	in $A_1 \otimes A_2,$
	define
	$\Lambda  (x) \colon \Omega \to A_1$
	by
	\[
		\Lambda (x) (\omega) = \sum_{i=1}^{n} x_{2,i}(\omega) x_{1,i}.
	\]
	The space $C(\Omega, A_1$ of continuous functions taking values in $A_1$ 
	is equipped with the $\text{C}^\ast$\nobreakdash-algebra structure
\begin{gather*}
	\lVert a \rVert = \sup\{ \lvert a(\omega) \rvert \colon 
	\omega \in \Omega \} \\
	(\lambda a + \mu b)(\omega) = \lambda a(\omega) + \mu b(\omega) \\
	(ab) (\omega) = a(\omega) b(\omega) \\
	a^\ast (\omega) = (a(\omega))^\ast
\end{gather*}
for $a,b$ in $C(\Omega, A_1),$ $\lambda, \mu$ in $\mathbb{C}$
and $\omega$ in $\Omega.$
For $x = \sum_{i=1}^{n} x_{1,i} \otimes x_{2,i}$ in $A_1 \otimes A_2,$
we have
\begin{align*}
	\lVert \Lambda (x) \rVert
	={}& \sup \{ \lVert \sum_{i=1}^{n} x_{2,i}(\omega) x_{1,i} \rVert
	\colon \omega \in \Omega \} \\
	={}& \sup \{ \lvert \sum_{i=1}^{n} x_{2,i}(\omega) \phi(x_{1,i}) \rvert
	\colon \phi \in A_1^\ast, \, \lVert \phi \rVert \leq 1, \, \omega \in \Omega \} \\
	={}& \sup \{ \lVert \Lambda( \sum_{i=1}^{n} \phi(x_{1,i}) x_{2,i})  \rVert
	\colon \phi \in A_1^\ast, \, \lVert \phi \rVert \leq 1, \, \omega \in \Omega \} \\
	={}& \sup \{ \lvert  \sum_{i=1}^{n} \phi(x_{1,i}) \psi( x_{2,i})  \rvert
	\colon \phi \in A_1^\ast, \, \lVert \phi \rVert \leq 1 \} \\
	={}& \sup \{ \lvert   \phi \otimes \psi ( \sum_{i=1}^{n} x_{1,i} \otimes x_{2,i})  \rvert
	\colon \phi \in A_1^\ast, \, \lVert \phi \rVert \leq 1 \,
\psi \in A_2^\ast, \, \lVert \psi \rVert \leq 1 \} \\
	={}& \lVert x \rVert_{\lambda} .
\end{align*}
Regard each $\omega$ as a one dimensional representation of $A_2$
and let $\pi_1$ be a faithful representation of $A_1.$
Then
\begin{align*}
	{}& \lVert x \rVert_\Lambda \\
	={}& \sup \{ \lVert \sum_{i=1}^{n} x_{2,i}(\omega) x_{1,i} \rVert
	\colon \omega \in \Omega \} \\
	={}& \sup \{ \lVert \pi_1( \sum_{i=1}^{n} x_{2,i}(\omega) x_{1,i} ) \rVert
	\colon \omega \in \Omega \} \\
	={}& \sup \{ \lVert (\pi_1 \otimes \pi_{\omega}) (\sum_{i=1}^{n} x_{1,i} \otimes x_{2,i} ) \rVert
	\colon \omega \in \Omega \} \\
	={}& \lVert (\pi_1 \otimes \oplus_{\omega \in \Omega} \pi_{\omega} )
	(\sum_{i=1}^{n} x_{1,i} \otimes x_{2,i} \rVert \\
	={}& \lVert x \rVert_{\min}.
\end{align*}
\end{proof}

	Then,

	{}& \lVert x \rVert_{\beta} \\

		\colon \omega_1 \in P(A_1)\, \omega_2 \in P(A_2) \} \\
	\leq{}& \sup\{ \lVert \pi_1 \otimes \pi_2 (x) \rVert \colon 
	\begin{array}{c}
		\pi_1 \text{ is some representation of } A_1 \\ 
		\pi_2 \text{ is some representation of } A_2
	\end{array} \} \\
	={}& \lVert x \rVert_{\min} .
\end{align*}
	Now we will prove that $S_{\beta}$ is all of $P(A_1) \times P(A_2).$
	Towards contradiction, suppose this is not the case. Then there are
	open sets $U_1,$ repsectively $U_2$ of $P(A_1),$ respectively $P(A_2)$
such that $U_1 \times U_2$ does not intersect $S_{\beta}.$ Replacing
$U_1$ by
\[
	\bigcup_{u \in \mathcal{U}(A_1)} u U_1 u^\ast
\]
and
$U_2$ by
\[
	\bigcup_{v \in \mathcal{U}(A_1)}  vU_2 v^\ast,
\]
we get that $U_1$ and $U_2$ are unitarily invariant and their product still
does not intersect $S_{\beta},$ by Lemma~\ref{lem:s_beta_unitarily_invariant}. This means
$K_1 = U_1^c$ and $K_2 = U_2^c$ are closed subsets of $P(A_1)$ and $P(A_2)$
which satisfy $S_{\beta} \subseteq K_1 \times  P(A_2) \cup P(A_1) \times K_2.$
Since $K_1$ is closed,
\[
	I_{K_1} = \{ x_1 \in A_1 \colon \omega (x_1) = 0 \text{ for all }
	\omega \in K_1 \}
\]
is a closed ideal of $A_1.$ This is due to \cite[Theorem 4.15]{takesaki79}. This means there exists a nonzero positive element
$y_1$ in $I_{K_1}$ such that $\omega_1(y_1) = 0$ for all $\omega_1$ in $K_1.$
Similarly, there exist a nonzero $y_2$ with the corresponding property.
So $(\omega_1 \otimes \omega_2)(y_1 \otimes y_2) = 0$ for all $\omega_1 \otimes
\omega_2$ in $P(A_{\beta}).$ Since all pure states in $P(A_{\beta})$ are of this 
form, this owuld mean $y_1 \otimes y_2 = 0.$
So, for $x$ in $A_1 \otimes A_2,$ we have
\begin{align*}
	\lVert x \rVert_{\lambda}
	={}& \lVert x \rVert_{\min}  \\
	={}& \sup\{ \lVert (\pi_1 \otimes \pi_2)(x) \rVert
		\colon \begin{array}{c}
		\pi_1 \text{ is some irreducible representation of } A_1 \\ 
		\pi_2 \text{ is some irreducible representation of } A_2
	\end{array} \} \\
		\leq{}& \lVert x \rVert_{\beta}.
\end{align*}
\end{proof}
